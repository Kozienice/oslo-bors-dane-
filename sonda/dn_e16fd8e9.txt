<script>
  (function () {
    if (window.__nhstHeaderTrackingInitialized) return;
    if (!window.NhstTracking) return; // bundle failed to load — fail quietly
    window.__nhstHeaderTrackingInitialized = true;

    /*
     * Capture the bundle instance WE initialize, and use only this reference when
     * firing. The paywall template (sub-paywall-templates) injects its own copy of
     * tracking.iife.js when a wall is shown; that re-evaluation OVERWRITES
     * window.NhstTracking with a fresh instance whose config never got an apiKey
     * (its init early-returns on the shared __nhstTrackingReady flag), so events
     * fired through it reach the eventContext data layer but are never sent to the
     * data platform. Because our fire is held for the Zephr outcome, it can land
     * after that overwrite on exactly the walled (logged-out) page views.
     */
    var trackingApi = window.NhstTracking;

    function detectPageType() {
      // Remove query params and trailing slashes.
      var path = (window.location.pathname || '').split('?')[0].replace(/\/$/, '');
      // Smak search pages (wine/recipe/restaurant) — listed explicitly as page.type = search.
      if (/\/smak\/(vinsok|oppskriftersok|restaurantsok)/.test(path)) return 'search';
      if (/(\d+)-(\d+)-(\w+)$/.test(path)) return 'article';
      return 'section';
    }

    /*
     * The host page (dn-webapp / dn-feature-webapp / dn-common-webapp) exposes a frozen
     * window.dn.datalayer with `article` + `issue` objects on article pages. The contract
     * requires article.id + article.title when the object is present; extra fields ride in
     * article.properties with key names/types matching the cpp producer contract v2. Only
     * sourceable, non-empty values are sent — on pages without a datalayer article the
     * article key is omitted entirely (contract: omit or null both mean "no article").
     */
    function addProp(props, key, value) {
      if (value === undefined || value === null || value === '') return;
      if (Array.isArray(value) && value.length === 0) return;
      props[key] = value;
    }

    function buildArticle(dl) {
      if (!dl || !dl.article) return null;
      var a = dl.article;
      var props = {};
      var publishTimestamp = Date.parse(a.publishDate);
      if (!isNaN(publishTimestamp)) props.publish_timestamp = publishTimestamp;
      addProp(props, 'category', a.category && a.category.default);
      addProp(props, 'topic', a.topic);
      addProp(props, 'tags', a.tags);
      addProp(props, 'authors', a.authors);
      if (dl.meta) {
        addProp(props, 'lifespan_score', dl.meta.levetid);
        addProp(props, 'news_score', dl.meta.nyhetsverdi);
      }
      var article = { id: a.id || null, title: a.originalTitle || null };
      if (Object.keys(props).length > 0) article.properties = props;
      return article;
    }

    /*
     * availability/wall_type derive from issue.release (stable: never mutated client-side;
     * wall_type = the kind of wall protecting the content, cpp semantics). wall = whether
     * the reader actually got walled: the Zephr outcome when it resolves within the hold
     * window (see fireWhenOutcomeKnown), else the server-rendered issue.extent. External
     * values are matched case-insensitively.
     */
    function buildPage(dl, wallOverride) {
      var page = { type: detectPageType() };
      var issue = dl && dl.issue;
      if (issue) {
        var release = typeof issue.release === 'string' ? issue.release.toLowerCase() : '';
        var extent = typeof issue.extent === 'string' ? issue.extent.toLowerCase() : '';
        page.availability = release === 'plus' ? 'plus' : 'free';
        page.wall = typeof wallOverride === 'boolean' ? wallOverride : extent === 'teaser';
        page.wall_type = release === 'plus' ? 'paywall' : 'open';
      }
      return page;
    }

    /*
     * The reader's REAL paywall outcome lives in window.Zephr.outcomes['dn-article'],
     * resolved asynchronously by the Zephr SDK. Mirrors the mapping the subscriptions
     * team's analytics-related-work.js uses: outcomeLabel 'show content' → no wall,
     * any other non-empty label → walled. Returns null while unresolved — including an
     * empty outcomeLabel, which their script also treats as "nothing to report".
     */
    function readZephrWall() {
      try {
        var oc = window.Zephr && window.Zephr.outcomes && window.Zephr.outcomes['dn-article'];
        var label = oc && oc.outcomeLabel ? String(oc.outcomeLabel) : '';
        if (!label) return null;
        return label.toLowerCase() !== 'show content';
      } catch (e) {
        return null;
      }
    }

    function firePageView(wallOverride) {
      var dl = window.dn && window.dn.datalayer;
      var api = trackingApi;
      /*
       * Hack: trackPageLoad applies its context twice (registered page context + the event),
       * and the library's merge concatenates arrays — so passing article there doubles
       * article.properties.authors/tags. Route article through updateEventDefaults instead:
       * it lands in rootTrackingContext (a single merge layer), so the arrays enter the
       * merge exactly once. article still rides every event (incl. this page_view) via the
       * defaults layer, and persists as page context for later click/scroll events.
       */
      var article = buildArticle(dl);
      if (article && typeof api.updateEventDefaults === 'function') {
        api.updateEventDefaults({ article: article });
      }
      api.trackPageLoad({ page: buildPage(dl, wallOverride) });
    }

    /*
     * On article pages the server-rendered issue.extent is only the server's assumption;
     * the actual outcome (walled or not) is decided client-side by Zephr, asynchronously.
     * Hold the page_view until the outcome resolves (poll every 250ms, cap 4s) so it
     * carries the truth; on timeout or when Zephr never appears, fire with the server
     * values. Non-article pages (no issue) fire immediately, unchanged.
     */
    function fireWhenOutcomeKnown() {
      var dl = window.dn && window.dn.datalayer;
      if (!dl || !dl.issue) {
        firePageView(null);
        return;
      }
      var wall = readZephrWall();
      if (wall !== null) {
        firePageView(wall);
        return;
      }
      var waited = 0;
      var timer = setInterval(function () {
        waited += 250;
        var w = readZephrWall();
        if (w !== null || waited >= 4000) {
          clearInterval(timer);
          firePageView(w);
        }
      }, 250);
    }

    trackingApi.initializeTracking({
      applicationName: 'dn-widget-webapp',
      applicationVersion: '1.0.0',
      enableActiveTimeTracking: true,
      producerContract: { name: 'dn_widget_webapp', version: '1', owner_type: 'application' }
    });

    /*
     * The host sets dn.datalayer in an inline script that appears AFTER this fragment in
     * every host page's markup — wait until the document has finished parsing so that
     * script has definitely run before we read the datalayer.
     */
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', fireWhenOutcomeKnown);
    } else {
      fireWhenOutcomeKnown();
    }
  })();
</script>
